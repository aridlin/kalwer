cmake_minimum_required(VERSION 3.22)
project(kalwer_games LANGUAGES C CXX)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../..")
set(CMAKE_CXX_STANDARD 20)
add_library(kalwer_games SHARED games_jni.cpp)
target_include_directories(kalwer_games PRIVATE "${ROOT}")
target_compile_options(kalwer_games PRIVATE -O2 -Wall -Wextra)
target_link_libraries(kalwer_games GLESv3 jnigraphics android log)
# The GPL Doom runtime remains a separate executable process. Name it as a
# native library so Android installs it into the executable APK library area.
file(STRINGS "${ROOT}/vendor/doomgeneric/Makefile" DOOM_LINE REGEX "^SRC_DOOM = ")
string(REPLACE "SRC_DOOM = " "" DOOM_LINE "${DOOM_LINE}")
string(REPLACE "doomgeneric_xlib.o" "" DOOM_LINE "${DOOM_LINE}")
separate_arguments(DOOM_OBJECTS UNIX_COMMAND "${DOOM_LINE}")
set(DOOM_SOURCES)
foreach(OBJECT IN LISTS DOOM_OBJECTS)
 string(REPLACE ".o" ".c" SOURCE "${OBJECT}")
 list(APPEND DOOM_SOURCES "${ROOT}/vendor/doomgeneric/${SOURCE}")
endforeach()
add_executable(koom_android ${DOOM_SOURCES} "${ROOT}/vendor/doomgeneric/mus2mid.c" "${ROOT}/koom/host.c" "${ROOT}/koom/audio.c")
target_include_directories(koom_android PRIVATE "${ROOT}/vendor/doomgeneric")
target_compile_definitions(koom_android PRIVATE NORMALUNIX _DEFAULT_SOURCE FEATURE_SOUND DOOMGENERIC_RESX=320 DOOMGENERIC_RESY=200)
target_compile_options(koom_android PRIVATE -O2)
target_link_libraries(koom_android m dl log OpenSLES)
set_target_properties(koom_android PROPERTIES OUTPUT_NAME libkalwer-koom SUFFIX .so RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/../../../build/generated/koom/${ANDROID_ABI}")
add_dependencies(kalwer_games koom_android)
